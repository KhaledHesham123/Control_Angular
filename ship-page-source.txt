import { Component, inject } from '@angular/core';
import { ShipBoardService } from './ship-board.service';
import { MeterFilter, PowerSource, ShipMode, ShipTank, TankCategory } from './ship.model';

@Component({
  selector: 'app-ship-overview',
  standalone: false,
  templateUrl: './ship-overview.html',
  styleUrls: ['./ship-board.css', './ship-twin-dashboard.css'],
})
export class ShipOverview {
  readonly board = inject(ShipBoardService);
  readonly radialLength = 2 * Math.PI * 26;
  readonly today = new Date().toLocaleDateString('ar-EG', {
    weekday: 'long',
    year: 'numeric',
    month: 'long',
    day: 'numeric',
  });

  meterFilter: MeterFilter = 'food';
  selectedZone = 'Bridge';

  readonly gauges: { emoji: string; label: string; where: string; unit: string; category: TankCategory; color: string }[] = [
    { emoji: '🔥', label: 'بنزين السفينة', where: 'تانكات الـ FUEL في قاع السفينة', unit: 'طن', category: 'hfo', color: '#f59e0b' },
    { emoji: '⚡', label: 'ديزل السفينة', where: 'تانكات الديزل في قاع السفينة', unit: 'طن', category: 'mgo', color: '#0f6a56' },
    { emoji: '🛢️', label: 'زيت المحركات', where: 'تانكات الزيت في غرفة المحركات', unit: 'طن', category: 'lube', color: '#eab308' },
    { emoji: '💧', label: 'مياه الشرب', where: 'تانكات المياه العذبة', unit: 'طن', category: 'water', color: '#1d6f8a' },
    { emoji: '🍱', label: 'الطعام والمؤن', where: 'مخازن الإعاشة والمطبخ', unit: 'كيلوجرام', category: 'food', color: '#9a3412' },
  ];

  readonly zones = [
    { id: 'Bridge', label: 'غرفة القيادة', longLabel: 'غرفة القيادة والإعاشة (Bridge)', color: '#38bdf8' },
    { id: 'Engine_Room', label: 'غرفة المحركات', longLabel: 'غرفة المحركات والمولدات والغلاية', color: '#22c55e' },
    { id: 'Tanks', label: 'التانكات', longLabel: 'منطقة التانكات والصابورة والمياه', color: '#f59e0b' },
    { id: 'Cargo_Hold_1', label: 'عنابر الشحن', longLabel: 'عنابر الشحن والمؤن (Cargo Holds)', color: '#ef4444' },
    { id: 'Cranes', label: 'الأوناش', longLabel: 'الأوناش ومعدات السطح (Deck Cranes)', color: '#eab308' },
  ];

  readonly meterFilters: { id: MeterFilter; label: string }[] = [
    { id: 'all', label: 'كل التانكات الظاهرة' },
    { id: 'hfo', label: 'FUEL' },
    { id: 'mgo', label: 'ديزل وزيت' },
    { id: 'water', label: 'مياه وصابورة' },
    { id: 'food', label: 'طعام ومؤن' },
    { id: 'cargo', label: 'عنبر الشحن' },
  ];

  setMode(mode: ShipMode): void {
    this.board.setMode(mode);
  }

  setPower(power: PowerSource): void {
    this.board.setPower(power);
  }

  mainGenerators(): number {
    return this.board.config().generators.filter((item) => item.role === 'main').length;
  }

  standbyGenerators(): number {
    return this.board.config().generators.filter((item) => item.role !== 'main').length;
  }

  previewTanks(): ShipTank[] {
    return this.board.tanks().filter((item) => {
      if (this.meterFilter === 'all') return item.category !== 'cargo' && item.category !== 'ballast';
      if (this.meterFilter === 'mgo') return item.category === 'mgo' || item.category === 'lube';
      if (this.meterFilter === 'water') return item.category === 'water' || item.category === 'ballast';
      return item.category === this.meterFilter;
    });
  }

  tankUnit(tank: ShipTank): string {
    return tank.unit === 'kg' ? 'كيلوجرام' : 'طن';
  }

  percent(item: ShipTank): number {
    return Math.round((item.current / item.max) * 100);
  }

  barColor(category: TankCategory): string {
    return this.gauges.find((item) => item.category === category)?.color ?? '#38bdf8';
  }

  daysBadgeClass(days: number): string {
    if (days < 5) return 'days-badge danger';
    if (days < 12) return 'days-badge warn';
    return 'days-badge';
  }

  radialOffset(percent: number): number {
    return this.radialLength * (1 - Math.max(0, Math.min(100, percent)) / 100);
  }

  selectZone(id: string): void {
    this.selectedZone = id;
    if (id === 'Cargo_Hold_1') this.meterFilter = 'cargo';
    else if (id === 'Bridge') this.meterFilter = 'food';
    else this.meterFilter = 'all';
  }

  twinNote(): string {
    if (this.selectedZone === 'Cranes') return 'Gearless: مفيش أوناش على السفينة، التحميل بأوناش الميناء فوق العنبرين.';
    if (this.selectedZone === 'Cargo_Hold_1') return 'عنبر واحد 52 × 12.6 × 8.275 م، فتحتان McGregor 25.15 × 12.57 م.';
    if (this.selectedZone === 'Engine_Room') return 'MAK 6M551AK · 1470/2647 kW · 425 RPM · IF 380.';
    if (this.selectedZone === 'Tanks') return 'تانكات القاع المزدوج تحت العنابر.';
    if (this.selectedZone === 'Bridge') return 'الإقامة وغرفة القيادة في المؤخرة. بيانات هذا الجزء: مخازن الإعاشة والطعام.';
    return 'MPP / Conbulker · فريتاون · 96.96 × 17.60 × غاطس 6.83 م';
  }

  zoneTitle(): string {
    const zone = this.zones.find((item) => item.id === this.selectedZone);
    return zone?.longLabel ?? 'السفينة';
  }
}

import { Component, DestroyRef, OnInit, inject, signal } from '@angular/core';
import { takeUntilDestroyed } from '@angular/core/rxjs-interop';
import { Title } from '@angular/platform-browser';
import { NavigationEnd, Router } from '@angular/router';
import { filter } from 'rxjs';
import { ShipBoardService } from './ship-board.service';
import { SHIP_NAV, SHIP_ROLES, ShipNavSection, ShipRole, ShipTank } from './ship.model';

interface BunkerSupplier {
  company: string;
  port: string;
  prices: string;
  supply: string;
  subtype: string;
}

interface MateRow {
  label: string;
  value: string;
}

interface MatePanel {
  kicker: string;
  title: string;
  route: string;
  link: string;
  rows: MateRow[];
}

@Component({
  selector: 'app-ship-shell',
  standalone: false,
  templateUrl: './ship-shell.html',
  styleUrls: ['./ship-board.css', './ship-twin-dashboard.css'],
})
export class ShipShell implements OnInit {
  readonly board = inject(ShipBoardService);
  readonly roles = SHIP_ROLES;
  readonly nav = SHIP_NAV;
  private readonly title = inject(Title);
  private readonly router = inject(Router);
  private readonly destroyRef = inject(DestroyRef);

  menuOpen = false;
  readonly path = signal(this.router.url.split('?')[0]);
  bunkerOpen = false;
  bunkerTankId = '';
  bunkerQty = 40;
  bunkerSupply = 'Fuel HFO';
  bunkerSubtype = 'VLSFO 0.5% / HFO 380';
  bunkerSupplierPick = '0';
  bunkerSupplier = 'مصر للبترول للتموين البحري (Misr Petroleum Marine)';
  bunkerPort = 'ميناء الإسكندرية / دمياط / السويس - مصر';
  bunkerInvoice = 'BNK-2026-205';
  bunkerCost = 32500;
  bunkerDate = new Date().toISOString().slice(0, 10);
  bunkerPrices = 'VLSFO: $545/MT | MGO: $785/MT';

  readonly bunkerSuppliers: BunkerSupplier[] = [
    { company: 'مصر للبترول للتموين البحري (Misr Petroleum Marine)', port: 'ميناء الإسكندرية / دمياط / السويس - مصر', prices: 'VLSFO: $545/MT | MGO: $785/MT', supply: 'Fuel HFO', subtype: 'VLSFO 0.5% / HFO 380' },
    { company: 'Minerva Bunkering Med', port: 'ميناء بيرايوس - اليونان', prices: 'HFO 380: $495/MT | MGO DMA: $760/MT', supply: 'Fuel HFO', subtype: 'HFO 380' },
    { company: 'Shell / Vivo Marine Bunkers', port: 'ميناء روتردام - هولندا', prices: 'Cylinder Oil SAE50: $2,150/MT | VLSFO: $520/MT', supply: 'Lube Oil', subtype: 'Cylinder Oil SAE50' },
    { company: 'الأهرام لتموين السفن (Al-Ahram Ship Chandlers)', port: 'ميناء دمياط والإسكندرية - مصر', prices: 'تموين غذائي شامل جاف ومبرد', supply: 'Provisions', subtype: 'مؤن جافة ومجمدات' },
    { company: 'التوكيلات الملاحية المصرية وقناة السويس', port: 'ميناء بورسعيد / دمياط - مصر', prices: 'مياه عذبة: $5.5/طن', supply: 'Fresh Water', subtype: 'مياه شرب عذبة' },
  ];

  ngOnInit(): void {
    this.title.setTitle('لوحة تحكم السفينة | Ship Dashboard');
    this.syncBunker(this.router.url);
    this.router.events.pipe(
      filter((event): event is NavigationEnd => event instanceof NavigationEnd),
      takeUntilDestroyed(this.destroyRef),
    ).subscribe((event) => this.syncBunker(event.urlAfterRedirects));
  }

  visibleSections(): ShipNavSection[] {
    const role = this.board.role();
    return this.nav
      .filter((section) => section.roles.includes(role))
      .map((section) => ({
        ...section,
        items: section.items.filter((item) => item.roles.includes(role)),
      }))
      .filter((section) => section.items.length > 0);
  }

  onRole(value: string): void {
    this.board.setRole(value as ShipRole);
  }

  supplyTanks(): ShipTank[] {
    return this.board.tanks().filter((item) => item.category !== 'cargo');
  }

  tankCaption(tank: ShipTank): string {
    const names: Record<string, string> = {
      hfo1: 'تانك بنزين 1 · FUEL',
      hfo2: 'تانك بنزين 2 · FUEL',
      hfo3: 'تانك بنزين 3 · FUEL',
      hfo4: 'تانك خدمة البنزين · FUEL',
      mgo1: 'تانك ديزل رقم 1',
      mgo2: 'تانك ديزل رقم 2',
      mgo3: 'تانك ديزل رقم 3',
      lube1: 'تانك الزيت الرئيسي',
      lube2: 'تانك الزيت الاحتياطي',
      fw1: 'تانك مياه الشرب 1',
      fw2: 'تانك مياه الشرب 2',
      wbt1: 'تانك صابورة 1',
      wbt2: 'تانك صابورة 2',
      food1: 'مخزن المؤن الجافة',
      food2: 'مخزن المجمدات',
      food3: 'خضروات وألبان',
    };
    const unit = tank.unit === 'kg' ? 'كجم' : 'طن';
    return `${names[tank.id] ?? tank.name} (الحالي: ${tank.current} / الأقصى: ${tank.max} ${unit})`;
  }

  selectedTank(): ShipTank | undefined {
    return this.supplyTanks().find((item) => item.id === this.bunkerTankId) ?? this.supplyTanks()[0];
  }

  unitLabel(): string {
    return this.selectedTank()?.unit === 'kg' ? 'كجم' : 'طن';
  }

  afterBunker(): string {
    const tank = this.selectedTank();
    if (!tank) return '';
    const next = Math.min(tank.max, tank.current + (Number(this.bunkerQty) || 0));
    return `الرصيد بعد التزويد سيصبح: ${next.toFixed(1)} من ${tank.max} ${this.unitLabel()}`;
  }

  onTankChange(): void {
    const tank = this.selectedTank();
    if (!tank) return;
    const room = Math.max(0, tank.max - tank.current);
    this.bunkerQty = Math.min(tank.category === 'food' ? 200 : 40, room || 10);
    if (tank.category === 'hfo') {
      this.bunkerSupply = 'Fuel HFO';
      this.bunkerSubtype = 'VLSFO 0.5% / HFO 380';
    } else if (tank.category === 'mgo') {
      this.bunkerSupply = 'Fuel MGO';
      this.bunkerSubtype = 'MGO DMA';
    } else if (tank.category === 'lube') {
      this.bunkerSupply = 'Lube Oil';
      this.bunkerSubtype = 'Cylinder Oil SAE50';
    } else if (tank.category === 'water' || tank.category === 'ballast') {
      this.bunkerSupply = 'Fresh Water';
      this.bunkerSubtype = 'مياه شرب عذبة';
    } else {
      this.bunkerSupply = 'Provisions';
      this.bunkerSubtype = 'مؤن جافة ومجمدات';
    }
  }

  onSupplierChange(): void {
    const supplier = this.bunkerSuppliers[Number(this.bunkerSupplierPick)];
    if (!supplier) return;
    this.bunkerSupplier = supplier.company;
    this.bunkerPort = supplier.port;
    this.bunkerPrices = supplier.prices;
    this.bunkerSupply = supplier.supply;
    this.bunkerSubtype = supplier.subtype;
  }

  saveBunker(): void {
    const tank = this.selectedTank();
    if (!tank) return;
    this.board.bunkering(tank.id, Number(this.bunkerQty) || 0, this.bunkerSupplier, this.bunkerPort);
    this.closeBunker();
  }

  closeBunker(): void {
    this.bunkerOpen = false;
    if (this.router.url.includes('/bunkering')) {
      void this.router.navigateByUrl('/ship');
    }
  }

    const freight = voyages.reduce((sum, item) => sum + (item.freightRevenue || 0), 0);
    const lowTanks = this.board.tanks().filter((tank) => tank.max > 0 && tank.current / tank.max < 0.35).length;
    const money = new Intl.NumberFormat('en-US', { maximumFractionDigits: 0 }).format(freight);
    const specs: MateRow[] = [
      { label: 'النوع', value: ship.type },
      { label: 'العلم', value: ship.flag },
      { label: 'الحمولة الساكنة', value: `${ship.dwt.toLocaleString('en-US')} MT` },

    ];
    const voyageRows: MateRow[] = [
      { label: 'الرحلات', value: String(voyages.length) },
      { label: 'محطات الموانئ', value: String(this.board.portCalls().length) },
      { label: 'نولون مسجّل', value: `$${money}` },
    ];
    const panels: Record<string, MatePanel> = {
      '/ship/fleet': { kicker: 'مرتبط بالأسطول', title: 'مواصفات السفينة المختارة', route: '/ship/particulars', link: 'فتح بطاقات المواصفات', rows: specs },
      '/ship/particulars': { kicker: 'مرتبط بالمواصفات', title: 'دليل الأسطول', route: '/ship/fleet', link: 'فتح كل السفن', rows: [{ label: 'السفن المسجلة', value: String(this.board.fleet().length) }, { label: 'السفينة الحالية', value: ship.name }, { label: 'IMO', value: ship.imo }] },
      '/ship/voyage': { kicker: 'مرتبط بالرحلة', title: 'العملاء والنولون', route: '/ship/clients', link: 'فتح العملاء والشحنات', rows: voyageRows },
      '/ship/route-map': { kicker: 'مرتبط بالمسار', title: 'الرحلات وحسابات الموانئ', route: '/ship/voyage', link: 'فتح الرحلات', rows: voyageRows },
      '/ship/clients': { kicker: 'مرتبط بالشحنة', title: 'الرحلات وخط السير', route: '/ship/voyage', link: 'فتح الرحلات', rows: voyageRows },
      '/ship/cargo': { kicker: 'مرتبط بالعنابر', title: 'الرحلات والحمولة', route: '/ship/voyage', link: 'فتح الرحلات', rows: voyageRows },

      '/ship/crew': { kicker: 'مرتبط بالطاقم', title: 'الشهادات والتأمين', route: '/ship/certificates', link: 'فتح الشهادات', rows: [{ label: 'أفراد الطاقم', value: String(this.board.config().crewCount) }, { label: 'الشهادات', value: String(this.board.certificates().length) }] },

===== ship-board.service.ts =====

import { Injectable, computed, signal } from '@angular/core';
import {
  DailyLog,
  FLEET,
  GaugeSummary,
  PowerSource,
  ShipCertificate,
  ShipConfig,
  ShipMode,
  ShipProfile,
  ShipRole,
  ShipTank,
  ShipVoyageRecord,
  PortCall,
  TankCategory,
  defaultCertificates,
  defaultConfig,
  defaultPortCalls,
  defaultTanks,
  defaultVoyages,
} from './ship.model';

@Injectable({ providedIn: 'root' })
export class ShipBoardService {
  readonly fleet = signal<ShipProfile[]>(FLEET.map((ship) => ({ ...ship })));
  readonly vesselId = signal('11');
  readonly role = signal<ShipRole>('Admin');
  readonly tanks = signal<ShipTank[]>(defaultTanks());
  readonly config = signal<ShipConfig>(defaultConfig());
  readonly certificates = signal<ShipCertificate[]>(defaultCertificates());
  readonly flag = signal('SIERRA LEONE');
  readonly logs = signal<DailyLog[]>([]);
  readonly voyages = signal<ShipVoyageRecord[]>(defaultVoyages());
  readonly portCalls = signal<PortCall[]>(defaultPortCalls());
  readonly notice = signal<string | null>(null);

  readonly vessel = computed(
    () => this.fleet().find((item) => item.id === this.vesselId()) ?? this.fleet()[0],
  );

  selectVessel(id: string): void {
    this.vesselId.set(id);
    this.notice.set(null);
  }

  addShip(ship: Omit<ShipProfile, 'id'>): string {
    const id = String(Math.max(0, ...this.fleet().map((item) => Number(item.id) || 0)) + 1);
    this.fleet.update((items) => [...items, { ...ship, id }]);
    this.vesselId.set(id);
    this.notice.set('تم حفظ السفينة.');
    return id;
  }

  restoreFleet(): void {
    this.fleet.set(FLEET.map((ship) => ({ ...ship })));
    this.vesselId.set(FLEET[0].id);
    this.notice.set('تم تحديث بيانات الأسطول القياسية بنجاح');
  }

  setRole(role: ShipRole): void {
    this.role.set(role);
  }

  setMode(mode: ShipMode): void {
    this.config.update((config) => ({ ...config, mode }));
  }

  setPower(powerSupplyMode: PowerSource): void {
    this.config.update((config) => ({ ...config, powerSupplyMode }));
  }

  gauge(category: TankCategory): GaugeSummary {
    const tanks = this.tanks().filter((item) => item.category === category);
    const current = tanks.reduce((sum, item) => sum + item.current, 0);
    const max = tanks.reduce((sum, item) => sum + item.max, 0) || 1;
    const daily = this.dailyUse(category);
    const days = current / Math.max(daily, 0.05);
    return {
      current,
      max,
      percent: Math.max(0, Math.min(100, (current / max) * 100)),
      daily,
      days,
      unit: tanks[0]?.unit ?? '',
      note: this.dailyNote(category, daily),
    };
  }

  bunkering(tankId: string, quantity: number, supplier: string, port: string): void {
    const added = Math.max(0, quantity);
    this.tanks.update((items) =>
      items.map((item) =>
        item.id === tankId ? { ...item, current: Math.min(item.max, item.current + added) } : item,
      ),
    );
    this.notice.set(`تم حفظ عملية التزويد وزيادة رصيد التانك. ${supplier || ''} · ${port || ''}`);
  }

  saveBalances(values: Record<string, number>): void {
    this.tanks.update((items) =>
      items.map((item) => {
        const next = values[item.id];
        if (next == null || Number.isNaN(next)) {
          return item;
        }
        return { ...item, current: Math.max(0, Math.min(item.max, next)) };
      }),
    );
    this.notice.set('Tank balances saved.');
  }

  saveConfig(config: ShipConfig): void {
    this.config.set({
      ...config,
      crewCount: Math.max(1, config.crewCount),
      generators: config.generators.map((item) => ({ ...item })),
    });
    this.notice.set('Ship settings saved.');
  }

  submitReport(input: {
    date: string;
    location: string;
    mode: ShipMode;
    power: PowerSource;
    hoursMainHfo: number;
    hoursMainMgo: number;
    hoursBoiler: number;
    hoursLube: number;
    generatorHours: number[];
  }): DailyLog {
    const config = this.config();
    const genMgo = input.generatorHours.reduce(
      (sum, hours, index) => sum + hours * (config.generators[index]?.rateTonsPerHour ?? 0),
      0,
    );
    const hfo = input.hoursMainHfo * config.hfoTonsPerHourSea;
    const mgo =
      input.hoursMainMgo * config.meMgoTonsPerHourPort +
      genMgo +
      input.hoursBoiler * config.boilerMgoTonsPerHour;
    const lube = input.hoursLube * config.lubeTonsPerHour;
    const water = (config.crewCount * config.waterLitersPerPersonDay) / 1000 + input.hoursBoiler * 0.03;
    const food = config.crewCount * config.foodKgPerPersonDay;

    this.deduct('hfo', hfo);
    this.deduct('mgo', mgo);
    this.deduct('lube', lube);
    this.deduct('water', water);
    this.deduct('food', food);
    this.config.update((current) => ({ ...current, mode: input.mode, powerSupplyMode: input.power, location: input.location }));

    const log: DailyLog = { date: input.date, location: input.location, hfo, mgo, lube, water, food };
    this.logs.update((items) => [log, ...items]);
    this.notice.set('Daily report saved and consumption deducted.');
    return log;
  }

  saveVoyage(value: ShipVoyageRecord): void {
    const voyageNumber = value.voyageNumber.trim();
    const voyageTitle = value.voyageTitle.trim();
    if (!voyageNumber || !voyageTitle) {
      return;
    }
    const saved = { ...value, voyageNumber, voyageTitle };
    if (saved.voyageId) {
      this.voyages.update((items) => items.map((item) => (item.voyageId === saved.voyageId ? saved : item)));
      this.portCalls.update((items) =>
        items.map((item) => (item.voyageId === saved.voyageId ? { ...item, voyageNumber } : item)),
      );
    } else {
      const voyageId = Math.max(0, ...this.voyages().map((item) => item.voyageId)) + 1;
      this.voyages.update((items) => [{ ...saved, voyageId }, ...items]);
    }
    this.notice.set('تم حفظ الرحلة وخط السير.');
  }

  savePortCall(value: PortCall): void {
    const portName = value.portName.trim();
    if (!portName) {
      return;
    }
    const voyage = this.voyages().find((item) => item.voyageId === value.voyageId);
    const saved = { ...value, portName, voyageNumber: voyage?.voyageNumber ?? value.voyageNumber };
    if (saved.portCallId) {
      this.portCalls.update((items) => items.map((item) => (item.portCallId === saved.portCallId ? saved : item)));
    } else {
      const portCallId = Math.max(0, ...this.portCalls().map((item) => item.portCallId)) + 1;
      this.portCalls.update((items) => [...items, { ...saved, portCallId }]);
    }
    this.notice.set('تم حفظ محطة الميناء ورسوم الـ DA.');
  }

  deletePortCall(portCallId: number): void {
    this.portCalls.update((items) => items.filter((item) => item.portCallId !== portCallId));
    this.notice.set('تم حذف المحطة من خط السير.');
  }

  moveTo(lat: number, lng: number, location: string): void {
    this.config.update((config) => ({ ...config, latitude: lat, longitude: lng, location }));
    this.notice.set('Ship position updated.');
  }

  addCertificate(value: Omit<ShipCertificate, 'id'>): void {
    this.certificates.update((items) => [{ ...value, id: crypto.randomUUID() }, ...items]);
    this.notice.set('Certificate saved.');
  }

  setFlag(flag: string): void {
    this.flag.set(flag.trim() || this.flag());
    this.notice.set('Flag state updated.');
  }

  clearNotice(): void {
    this.notice.set(null);
  }

  private deduct(category: TankCategory, amount: number): void {
    let left = amount;
    this.tanks.update((items) =>
      items.map((item) => {
        if (item.category !== category || left <= 0 || item.current <= 0) {
          return item;
        }
        const used = Math.min(item.current, left);
        left -= used;
        return { ...item, current: item.current - used };
      }),
    );
  }

  private dailyUse(category: TankCategory): number {
    const config = this.config();
    const genDaily =
      config.powerSupplyMode === 'aux_gen'
        ? config.generators.reduce((sum, item) => sum + item.rateTonsPerHour * item.defaultHours, 0)
        : 0;
    if (category === 'hfo') {
      return config.mode === 'sea' ? config.hfoTonsPerHourSea * 24 : 0.1;
    }
    if (category === 'mgo') {
      const boiler = config.boilerMgoTonsPerHour * (config.mode === 'sea' ? config.boilerHoursSea : config.boilerHoursPort);
      const maneuver = config.mode === 'port' ? config.meMgoTonsPerHourPort * 24 : 0;
      return genDaily + boiler + maneuver;
    }
    if (category === 'lube') {
      return config.lubeTonsPerHour * (config.mode === 'sea' ? 24 : 12);
    }
    if (category === 'water') {
      return (config.crewCount * config.waterLitersPerPersonDay) / 1000 + 0.4;
    }
    if (category === 'food') {
      return config.crewCount * config.foodKgPerPersonDay;
    }
    return 1;
  }

  private dailyNote(category: TankCategory, daily: number): string {
    const unit = category === 'food' ? 'كيلوجرام' : 'طن';
    if (category === 'hfo' && this.config().mode === 'port') {
      return 'المحرك الرئيسي واقف في الميناء، والاستهلاك شبه متوقف';
    }
    return `${daily.toFixed(2)} ${unit}`;
  }
}
